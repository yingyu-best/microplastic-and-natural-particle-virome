#### mapping between virus and host genomes ####

#blastn between viral and bacterial genomes：

makeblastdb -in host_MAG.fna -dbtype nucl

blastn -query virus.fna \
       -db host_MAG.fna \
       -outfmt "6 qseqid sseqid pident length qlen slen qstart qend sstart send evalue" \
       -evalue 1e-5 \
       -out virus_vs_host.blast

{

}' A_S37C5056__A-TR12I-MP.31_k141_296432_length_5868_cov_45.6356.filtered.blast > comparison.tsv
